56/// Wrapper for channel priority registers
57///
58/// Channel priority registers cannot be accessed with
59/// normal channel indexes. This adapter makes it so that
60/// we *can* access them with channel indexes by converting
61/// the channel number to a reference to the priority
62/// register.
63#[repr(transparent)]
64pub struct ChannelPriorityRegisters([RWRegister<u8>; 32]);
65
66impl Index<usize> for ChannelPriorityRegisters {
67    type Output = RWRegister<u8>;
68    fn index(&self, channel: usize) -> &RWRegister<u8> {
69        // Pattern follows
70        //
71        //   3, 2, 1, 0, 7, 6, 5, 4, 11, 10, 9, 8, ...
72        //
73        // for all channels. NXP keeping us on our toes.
74        let idx = 4 * (channel / 4) + (3 - (channel % 4));
75        &self.0[idx]
76    }
77}
